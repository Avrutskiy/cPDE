cmake_minimum_required(VERSION 3.18)
project(cpde_server LANGUAGES CXX)
find_package(Threads REQUIRED)

set(CPDE_BACKEND "CUDA" CACHE STRING "Backends to build: CUDA, CPU, or BOTH")
set_property(CACHE CPDE_BACKEND PROPERTY STRINGS CUDA CPU BOTH)
string(TOUPPER "${CPDE_BACKEND}" CPDE_BACKEND)
if(NOT CPDE_BACKEND MATCHES "^(CUDA|CPU|BOTH)$")
    message(FATAL_ERROR "CPDE_BACKEND must be CUDA, CPU, or BOTH")
endif()
option(CPDE_ENABLE_NVRTC "Enable runtime compilation of CUDA scalar expressions" ON)
option(CPDE_BUILD_ENGINE_TESTS "Build engine storage regression probes" OFF)
option(CPDE_BUILD_REDUCTION_TESTS "Build reduction regression probes" OFF)
if(CPDE_BUILD_ENGINE_TESTS OR CPDE_BUILD_REDUCTION_TESTS)
    enable_testing()
endif()

function(cpde_cpp_target target)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
    target_link_libraries(${target} PRIVATE Threads::Threads)
endfunction()

if(CPDE_BACKEND STREQUAL "CUDA" OR CPDE_BACKEND STREQUAL "BOTH")
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_executable(cpde_server cuda/cpde_server.cu)
    cpde_cpp_target(cpde_server)
    set_target_properties(cpde_server PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    target_link_libraries(cpde_server PRIVATE CUDA::cublas)
    if(CPDE_ENABLE_NVRTC)
        target_compile_definitions(cpde_server PRIVATE CPDE_ENABLE_NVRTC)
        target_link_libraries(cpde_server PRIVATE CUDA::nvrtc CUDA::cuda_driver)
    endif()
    install(TARGETS cpde_server RUNTIME DESTINATION bin)
    if(CPDE_BUILD_ENGINE_TESTS)
        add_executable(cpde_engine_storage_probe tests/engine_storage_probe.cu)
        cpde_cpp_target(cpde_engine_storage_probe)
        target_include_directories(cpde_engine_storage_probe PRIVATE cuda)
        set_target_properties(cpde_engine_storage_probe PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
        target_link_libraries(cpde_engine_storage_probe PRIVATE CUDA::cublas)
        add_test(NAME cpde_engine_storage COMMAND cpde_engine_storage_probe)
    endif()
    if(CPDE_BUILD_REDUCTION_TESTS)
        add_executable(cpde_reduction_probe tests/reduction_probe.cu)
        cpde_cpp_target(cpde_reduction_probe)
        target_include_directories(cpde_reduction_probe PRIVATE cuda)
        set_target_properties(cpde_reduction_probe PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
        target_link_libraries(cpde_reduction_probe PRIVATE CUDA::cudart CUDA::cublas)
        add_test(NAME cpde_reductions COMMAND cpde_reduction_probe)
    endif()
endif()

if(CPDE_BACKEND STREQUAL "CPU" OR CPDE_BACKEND STREQUAL "BOTH")
    set(CPDE_CPU_BLAS "AUTO" CACHE STRING "CPU BLAS: AUTO, SYSTEM (required), or NONE (serial)")
    set_property(CACHE CPDE_CPU_BLAS PROPERTY STRINGS AUTO SYSTEM NONE)
    string(TOUPPER "${CPDE_CPU_BLAS}" CPDE_CPU_BLAS)
    if(NOT CPDE_CPU_BLAS MATCHES "^(AUTO|SYSTEM|NONE)$")
        message(FATAL_ERROR "CPDE_CPU_BLAS must be AUTO, SYSTEM, or NONE")
    endif()
    set(cpu_blas_libraries "")
    set(cpu_blas_definition "")
    if(NOT CPDE_CPU_BLAS STREQUAL "NONE")
        if(APPLE)
            find_library(CPDE_ACCELERATE Accelerate)
            if(CPDE_ACCELERATE)
                set(cpu_blas_libraries "${CPDE_ACCELERATE}")
                set(cpu_blas_definition CPDE_CPU_ACCELERATE)
                if(NOT CMAKE_OSX_DEPLOYMENT_TARGET OR NOT CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "13.3")
                    list(APPEND cpu_blas_definition ACCELERATE_NEW_LAPACK)
                endif()
            endif()
        else()
            find_package(BLAS QUIET)
            if(BLAS_FOUND)
                include(CheckCXXSourceCompiles)
                set(CMAKE_REQUIRED_LIBRARIES ${BLAS_LIBRARIES})
                check_cxx_source_compiles("#include <cblas.h>\nint main(){float x=1; return int(cblas_sasum(1,&x,1));}" CPDE_HAVE_CBLAS)
                unset(CMAKE_REQUIRED_LIBRARIES)
                if(CPDE_HAVE_CBLAS)
                    set(cpu_blas_libraries ${BLAS_LIBRARIES})
                    set(cpu_blas_definition CPDE_CPU_CBLAS)
                endif()
            endif()
        endif()
    endif()
    if(CPDE_CPU_BLAS STREQUAL "SYSTEM" AND NOT cpu_blas_definition)
        message(FATAL_ERROR "A system CBLAS library and headers are required; use CPDE_CPU_BLAS=AUTO or NONE for the portable fallback")
    endif()
    if(cpu_blas_definition)
        message(STATUS "cPDE CPU matrix operations: ${cpu_blas_definition}")
    else()
        message(STATUS "cPDE CPU matrix operations: portable serial implementation")
    endif()
    function(cpde_cpu_target target source)
        add_executable(${target} ${source})
        cpde_cpp_target(${target})
        target_include_directories(${target} PRIVATE cpu/include cuda)
        target_compile_definitions(${target} PRIVATE CPDE_CPU_BACKEND ${cpu_blas_definition})
        target_link_libraries(${target} PRIVATE ${cpu_blas_libraries})
    endfunction()
    cpde_cpu_target(cpde_cpu_server cpu/server.cpp)
    install(TARGETS cpde_cpu_server RUNTIME DESTINATION bin)
    if(CPDE_BUILD_ENGINE_TESTS)
        cpde_cpu_target(cpde_cpu_engine_storage_probe cpu/engine_storage_probe.cpp)
        add_test(NAME cpde_cpu_engine_storage COMMAND cpde_cpu_engine_storage_probe)
    endif()
    if(CPDE_BUILD_REDUCTION_TESTS)
        cpde_cpu_target(cpde_cpu_reduction_probe cpu/reduction_probe.cpp)
        add_test(NAME cpde_cpu_reductions COMMAND cpde_cpu_reduction_probe)
    endif()
endif()
